<script>

const bites=[
document.getElementById("bite1"),
document.getElementById("bite2"),
document.getElementById("bite3"),
document.getElementById("bite4")
];

let bite=0;

const cookie=document.getElementById("cookie");

cookie.onclick=()=>{

if(bite<bites.length){

bites[bite].setAttribute("r",28+Math.random()*6);

makeSparkles();

bite++;

}

if(bite===bites.length){

cookie.style.pointerEvents="none";

setTimeout(()=>{
document.getElementById("msg").classList.add("show");
celebrate();
},500);

}

};

function makeSparkles(){

for(let i=0;i<6;i++){

const s=document.createElement("div");

s.className="sparkle";

s.innerHTML=["✨","⭐","💛","🍪"][Math.floor(Math.random()*4)];

s.style.left=(window.innerWidth/2+(Math.random()*160-80))+"px";

s.style.top=(window.innerHeight/2+(Math.random()*80-40))+"px";

document.body.appendChild(s);

setTimeout(()=>s.remove(),1500);

}

}

function celebrate(){

for(let i=0;i<30;i++){

setTimeout(makeSparkles,i*80);

}

}
